function extractMeta(rows){  
let branch = '', fromD = null, toD = null;  
const branchPattern = /^([A-Za-z]+)\s*0*(\d+)\s*-\s*[A-Za-z]+$/;  
const datePattern = /From\s*Date\s*:?\s*(\d{1,2})[\/-](\d{1,2})[\/-](\d{4}).*?To\s*Date\s*:?\s*(\d{  
for(const row of rows){  
if(!row) continue;  
for(const cell of row){  
const text = (cell===undefined || cell===null) ? '' : String(cell);  
const trimmed = text.trim();  
if(!branch){  
const m = trimmed.match(branchPattern);  
if(m){  
branch = `${m[1].toUpperCase()}-${m[2]} Branch`;  
}
}
if(!fromD){  
const dm = text.match(datePattern);  
if(dm){  
const f = new Date(+dm[3], +dm[2]-1, +dm[1]);  
const t = new Date(+dm[6], +dm[5]-1, +dm[4]);  
if(!isNaN(f) && !isNaN(t)){  
fromD = f;  
toD = t;  
}
}
}
}
if(branch && fromD) break;  
}
return {branch, fromD, toD};  
}

function processRows(rows){

// Work out the column gap from "Order Type" to "Net Sales" and to "Cheque" —  
// that same gap holds for every data row's own Order Type value cell  
// (DINE-IN / Home Delivery / Take Away), even when the header/totals repeat on every line.  
let headerRowIdx = -1, netSalesOffset = null, chequeOffset = null;  
for(let i=0;i<rows.length;i++){  
const row = rows[i];  
const idx = row.findIndex(c => norm(c) === 'order type');  
if(idx !== -1){  
headerRowIdx = i;  
const netSalesCol = row.findIndex(c => norm(c) === 'net sales');  
const chequeCol = row.findIndex(c => norm(c) === 'cheque');  
if(netSalesCol !== -1) netSalesOffset = netSalesCol - idx;  
if(chequeCol !== -1) chequeOffset = chequeCol - idx;  
break;  
}
}
if(headerRowIdx === -1 || netSalesOffset === null || chequeOffset === null){  
setStatus('Could not find "Order Type", "Net Sales" and "Cheque" columns in this file. Please che  
return;  
}

{label:'Take Away', sale:0, trans:0, rows:0},  
sale:0, trans:0, rows:0},  
delivery:{label:'Delivery', sale:0, trans:0, rows:0}  
};  
let matchedRows = 0;  
rows.forEach((row, rIdx) => {  
if(!row || row.length===0) return;  
// Note: we deliberately do NOT skip the header row here — in some ERP exports  
// (like the repeated-header-per-line CSV) the very same row also carries the  
// first real data entry, so skipping it would silently drop that row.  
for(let idx=0; idx<row.length; idx++){  
const bucketKey = classify(row[idx]);  
if(!bucketKey) continue;  
const netVal = toNumber(row[idx + netSalesOffset]);  
const transVal = toNumber(row[idx + chequeOffset]);  
if(isNaN(netVal) || isNaN(transVal)) continue; // not a real hit at this offset, keep scanning  
buckets[bucketKey].sale += netVal;  
buckets[bucketKey].trans += transVal;  
buckets[bucketKey].rows += 1;

break; // one order-type value per row is enough  
}
});
